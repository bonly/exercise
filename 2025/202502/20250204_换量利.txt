{===== 1. 基础数据计算（训练模式保护） =====}
LB := V / MA(REF(V,1), 5); {今日量比}
HSL := IF(CAPITAL > 0, VOL / CAPITAL * 100, 0); {实时换手率}

LB_S := (LB - MA(LB, 5))/STD(LB,5);
HSL_S := (HSL -MA(HSL,5))/STD(HSL,5);

{ 用 WINNER 变化判断主力可能的操作 }
W_RAW := IF(TRAIN=1, 0, WINNER(C)*100); {当前获利盘比例，百分比}
W := IF(W_RAW > 0, W_RAW, 0);
W_C := IF(W > 0 AND REF(W,1) > 0, (W - REF(W,1)) / REF(W,1) * 100, 0); {获利盘比例变化百分比}

WN := WINNER(C)*100;
W_S := (WN - MA(WN, 20))/STD(WN, 20);   
{===== 2. MACD风格综合动能 =====}
MACD := LB_S + HSL_S; 

{===== 3. 绘制柱体 =====}
STICKLINE(MACD > 0, 0, MACD, 2, 0), COLORRED;
STICKLINE(MACD < 0, 0, MACD, 2, 0), COLORGREEN;
STICKLINE(MACD > 0 AND MACD < REF(MACD, 1), 0, MACD, 2, 0), COLORYELLOW; 
{===== 线 ====}
LB_S,COLORMAGENTA,LINETHICK1;
W_S, COLORYELLOW,LINETHICK1;

{===== 抬头表 ====}
MACD,COLORWHITE,NODRAW;
获利盘:WN,COLORYELLOW,NODRAW;
量比:LB,COLORMAGENTA,NODRAW;
换手率:HSL,NODRAW;
C1:COST(10),NODRAW;
C9:COST(90),NODRAW;
{
DIF, COLORYELLOW, LINETHICK2;
DEA, COLORMAGENTA, LINETHICK2;
}
{===== 4. 右侧数据区（绑定最新BAR） =====}
DRAWTEXT_FIX(ISLASTBAR, 0.50, 0.01, 0, STRCAT('LB:', CON2STR(LB, 2))), COLORLIBLUE;

{===== 5. 左侧信号区 =====}
{
DRAWTEXT_FIX(ISLASTBAR AND CROSS(DIF, DEA), 0.05, 0.05, 0, '金叉'), COLORRED;
DRAWTEXT_FIX(ISLASTBAR AND CROSS(DEA, DIF), 0.05, 0.20, 0, '死叉'), COLORGREEN;
DRAWTEXT_FIX(ISLASTBAR AND TRAIN=0 AND W_CHANGE < 0 AND LB > 1.5 AND HSL > 5 AND HSL < 10, 0.05, 0.35, 0, '共振'), COLORYELLOW;
DRAWTEXT_FIX(ISLASTBAR AND TRAIN=0 AND W_CHANGE > 0 AND LB > 2.5 AND HSL > 15, 0.05, 0.50, 0, '派发'), COLORGREEN;
}

0,COLORLIBLUE,LINETHICK1;
